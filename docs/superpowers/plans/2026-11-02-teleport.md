# Teleport Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 Scro Swap 下方加入獨立的 Teleport（`/app/teleport`），把原片裡的人放進新場景，留住臉、穿搭與動作。

**Architecture:** 新的 `teleport_jobs` 與 `src/service/teleport/`。純函式負責驗證、提示、秒數與任務標題。背景作業重用 Scro Swap 的抽幀、SAM 假人、Seedream、MiniMax H3、Hailuo 首尾幀、原聲混流與 H3 Soft。假人那一次 SAM 順便交出第一幀遮罩，用來做身份去背；沒有遮罩就用原始起始幀。`advanceTeleportJobs` 只輪詢已提交的影片或套用 25 分鐘逾時，不重跑準備步驟。

**Tech Stack:** Next.js Server Actions、MongoDB、現有 OpenRouter Seedream / Hailuo、fal SAM 3、MiniMax H3、FFmpeg、Node test runner（`npx tsx --test`）。

**Spec:** `docs/superpowers/specs/2026-11-02-teleport-design.md`

## Global Constraints

- 不在 `jutsu_jobs` 加模式，也不改 Scro Swap 的換角流程。
- 沒有角色選擇、17 款打光、遮罩邊距、帽子修正、口型開關、動作尺寸這些進階控制。
- 不扣 credits。
- 沒有可重複使用的場景庫。
- 成片比例只有 `9:16`。
- 不做「人留在原位、只換身後背景」的去背貼圖模式。
- 側欄在 Scro Swap 正下方。英文 `Teleport`，中文 `Teleport 換場`。icon 是現有的 `room`。
- 片段長度檔案記 2–15 秒（預設 6）。送給影片模型的 duration 是 `Math.min(15, Math.max(5, Math.round(sectionLength)))`。
- 場景照與場景文字至少要有一個。空白文字視同沒有文字。有照片時不另生成地點。只有文字時才生成沒有人的場景板，並把 `scenePlateGenerated` 設為 `true`。
- 假人固定 `margin: "normal"`、`motionSize: 512`、`hatFix: false`、`target: "person"`。
- H3 不送 `reference_audio`，不送 `first_frame` / `last_frame`，也不加上「嘴巴保持放鬆」那句。
- 身份去背只用 fal SAM 的第一幀二值遮罩。VLM 橢圓不算去背成功。失敗時 `identityFrameUrl` 用原始起始幀，影片提示要忽略那張圖的背景。
- 頁面每 4.5 秒輪詢。推進最多 12 筆。沒有 status URL 超過 25 分鐘：「Teleport 處理逾時，請重試」。有 status URL 仍在跑超過 25 分鐘：「Teleport 渲染逾時，請重試」。
- 任務標題是場景文字第一行，最多 40 個字；否則 `Teleport`。詳情 `tasksPage.detail.teleport`。`href` 是 `/app/teleport`。
- 刪除只刪自己的文件，不刪 Blob。列表 24 筆。
- 測試與驗收不打 Seedream、MiniMax、Hailuo、fal 或 Higgsfield。
- 同一實作回合更新 `pipeline-models.md` 與 `AGENTS.md`。
- 頁面元件放在 `src/presentation/components/app/teleport/`，各自一個檔案。有 `useEffect` 的檔案頂部加 `"use client"`。
- Server Action 只轉呼叫 service，不新增 REST route。

```ts
import { isBrandAssetUrl } from "@/service/video-edit/edit-state";

export type TeleportStartInput = {
  sourceVideoUrl: string;
  sectionStart?: number;
  sectionLength?: number;
  sceneImageUrl?: string;
  scenePrompt?: string;
  keepSourceAudio?: boolean;
  applyColorFix?: boolean;
};

export function normalizeTeleportStart(input: TeleportStartInput) {
  const sourceVideoUrl = input.sourceVideoUrl.trim();
  const sceneImageUrl = input.sceneImageUrl?.trim() || undefined;
  const scenePrompt = input.scenePrompt?.trim() || undefined;
  if (!sceneImageUrl && !scenePrompt) {
    return { ok: false as const, error: "請上傳場景照或寫下場景" };
  }
  const sectionStart = Math.max(0, Number(input.sectionStart) || 0);
  const sectionLength = Math.min(15, Math.max(2, Number(input.sectionLength) || 6));
  return {
    ok: true as const,
    sourceVideoUrl,
    sectionStart,
    sectionLength,
    sceneImageUrl,
    scenePrompt,
    keepSourceAudio: input.keepSourceAudio !== false,
    applyColorFix: input.applyColorFix !== false,
  };
}

/** Same allowlist as `startScroSwapJob`: https, and when a store host exists, brand folder or any public blob host. */
export function isAllowedTeleportMediaUrl(
  url: string,
  clerkUserId: string,
  storeHost: string | null,
): boolean {
  const trimmed = url.trim();
  if (!trimmed.startsWith("https://")) return false;
  if (!storeHost) return true;
  if (isBrandAssetUrl(trimmed, clerkUserId, storeHost)) return true;
  return trimmed.includes(".public.blob.vercel-storage.com/");
}

export function teleportSubmitDurationSeconds(sectionLength: number): number {
  return Math.min(15, Math.max(5, Math.round(sectionLength)));
}
```

```ts
export function teleportTaskTitle(scenePrompt?: string): string {
  const line = (scenePrompt ?? "").split(/\r?\n/)[0]?.trim() ?? "";
  if (!line) return "Teleport";
  return [...line].slice(0, 40).join("");
}

export function chooseTeleportVideoEngine(input: {
  hasMannequin: boolean;
  hasMinimaxKey: boolean;
}): "h3" | "hailuo" {
  return input.hasMannequin && input.hasMinimaxKey ? "h3" : "hailuo";
}

export function buildTeleportScenePlatePrompt(scenePrompt: string): string {
  return `Create a photoreal 9:16 scene still with no people, no faces, no bodies, and no mannequins. Location: ${scenePrompt.trim()}. The frame is empty of humans. Natural perspective and lighting.`;
}

export function buildTeleportRelocatePrompt(input: {
  moment: "start" | "end";
  scenePrompt?: string;
}): string {
  const mood = input.scenePrompt?.trim()
    ? `Apply this change to the place: ${input.scenePrompt.trim()}.`
    : "";
  return [
    "Attached image 1 is the person to keep. Attached image 2 is the place they must enter.",
    "Keep this exact person's face, body, outfit, expression, and the action of this instant.",
    "Place them inside attached image 2. Position, scale, and lighting may change so they belong in that place.",
    "Do not invent a different location. Do not keep the old background from attached image 1.",
    `This is the ${input.moment} moment.`,
    mood,
  ]
    .filter(Boolean)
    .join(" ");
}

export function buildTeleportVideoPrompt(input: {
  engine: "h3" | "hailuo";
  motionText: string;
  scenePrompt?: string;
  identityIgnoresBackground: boolean;
}): string {
  const mood = input.scenePrompt?.trim() ? input.scenePrompt.trim() : "";
  if (input.engine === "hailuo") {
    return [
      "[Shot 1] The shot begins from the attached first frame and transitions smoothly to the attached last frame.",
      "Keep the same person, outfit, and the new scene. Do not return to the original location.",
      input.motionText,
      mood,
    ]
      .filter(Boolean)
      .join(" ");
  }
  return [
    "Reference image 1 is the person already standing in the new scene. Keep that environment.",
    "Reference image 2 is the identity lock: keep that face, body, and outfit.",
    input.identityIgnoresBackground
      ? "Ignore the background of reference image 2."
      : "",
    "Reference video 1 is a grey mannequin. Follow its pose, hands, and timing.",
    input.motionText,
    mood,
  ]
    .filter(Boolean)
    .join(" ");
}
```

```ts
export async function teleportJobsCollection() {
  const db = await getDb();
  const collection = db.collection<OptionalId<TeleportJob>>("teleport_jobs");
  if (!indexed) {
    indexed = true;
    await Promise.all([
```

```ts
import sharp from "sharp";

/** Keep person pixels and paint every other pixel #f2f2f2. `mask` is row-major, 1 = person. */
export async function cutSubjectOntoPlain(
  frame: Buffer,
  mask: Uint8Array,
  width: number,
  height: number,
): Promise<Buffer> {
  if (mask.length !== width * height) {
    throw new Error("身份遮罩尺寸與畫格不符");
  }
  const { data } = await sharp(frame)
    .resize(width, height, { fit: "fill" })
    .ensureAlpha()
    .raw()
    .toBuffer({ resolveWithObject: true });
  for (let i = 0; i < width * height; i++) {
    if (mask[i]) continue;
    const o = i * 4;
    data[o] = 242;
    data[o + 1] = 242;
    data[o + 2] = 242;
    data[o + 3] = 255;
  }
  return sharp(data, { raw: { width, height, channels: 4 } })
    .webp({ quality: 88 })
    .toBuffer();
}
```
